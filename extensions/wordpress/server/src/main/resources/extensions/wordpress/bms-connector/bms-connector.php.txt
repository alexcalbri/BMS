const BMS_CONNECTOR_VERSION = '0.3.0';
const BMS_CONNECTOR_OPTION  = 'bms_connector_settings';
const BMS_CONNECTOR_QUEUE   = 'bms_connector_queue';
const BMS_CONNECTOR_CRON    = 'bms_connector_retry';
const BMS_CONNECTOR_SKEW    = 300;

/* ---------------------------------------------------------------------
 * Ajustes

function bms_connector_settings(): array {
	$saved = get_option( BMS_CONNECTOR_OPTION, [] );
	return wp_parse_args( is_array( $saved ) ? $saved : [], [
		'bms_url'            => '',
		'company'            => '',
		'secret'             => '',
		'send_views'         => '1',
		'recaptcha_enabled'  => '0',
		'recaptcha_site_key' => '',
		'recaptcha_secret'   => '',
		'recaptcha_score'    => '0.5',
	] );
}

function bms_connector_ready(): bool {
	$settings = bms_connector_settings();
	return $settings['bms_url'] !== '' && $settings['company'] !== '' && strlen( $settings['secret'] ) >= 32;
}

function bms_connector_sign( string $timestamp, string $body, string $secret ): string {
	return hash_hmac( 'sha256', $timestamp . '.' . $body, $secret );
}

function bms_connector_sanitize_settings( $input ): array {
	$current = bms_connector_settings();
	$input = is_array( $input ) ? $input : [];
	$secret = trim( (string) ( $input['secret'] ?? '' ) );
	$recaptcha_secret = trim( (string) ( $input['recaptcha_secret'] ?? '' ) );
	$score = (float) ( $input['recaptcha_score'] ?? 0.5 );
	return [
		'bms_url'            => untrailingslashit( esc_url_raw( trim( (string) ( $input['bms_url'] ?? '' ) ), [ 'https', 'http' ] ) ),
		'company'            => preg_replace( '/[^a-z0-9_]/', '', strtolower( (string) ( $input['company'] ?? '' ) ) ),
		// Un campo vacio conserva el secreto guardado.
		'secret'             => $secret !== '' ? preg_replace( '/[^A-Za-z0-9]/', '', $secret ) : $current['secret'],
		'send_views'         => empty( $input['send_views'] ) ? '0' : '1',
		'recaptcha_enabled'  => empty( $input['recaptcha_enabled'] ) ? '0' : '1',
		'recaptcha_site_key' => sanitize_text_field( (string) ( $input['recaptcha_site_key'] ?? '' ) ),
		'recaptcha_secret'   => $recaptcha_secret !== '' ? sanitize_text_field( $recaptcha_secret ) : $current['recaptcha_secret'],
		'recaptcha_score'    => (string) min( 1, max( 0, $score ) ),
	];
}

add_action( 'rest_api_init', static function () {
	register_rest_route( 'bms/v1', '/rpc', [
		'methods'             => 'POST',
		'callback'            => 'bms_connector_rpc',
		'permission_callback' => 'bms_connector_verify_request',
	] );
} );

function bms_connector_verify_request( WP_REST_Request $request ): bool {
	$settings = bms_connector_settings();
	$timestamp = (string) $request->get_header( 'X-BMS-Timestamp' );
	$signature = (string) $request->get_header( 'X-BMS-Signature' );
	if ( strlen( $settings['secret'] ) < 32 || ! ctype_digit( $timestamp ) || abs( time() - (int) $timestamp ) > BMS_CONNECTOR_SKEW ) {
		return false;
	}
	return hash_equals( bms_connector_sign( $timestamp, (string) $request->get_body(), $settings['secret'] ), strtolower( $signature ) );
}

function bms_connector_rpc( WP_REST_Request $request ): WP_REST_Response {
	$data = json_decode( (string) $request->get_body(), true );
	$data = is_array( $data ) ? $data : [];
	switch ( (string) ( $data['action'] ?? '' ) ) {
		case 'ping':
			return new WP_REST_Response( [ 'ok' => true, 'version' => BMS_CONNECTOR_VERSION, 'site' => get_bloginfo( 'name' ), 'theme' => wp_get_theme()->get( 'Name' ) ] );
		case 'search':
			return new WP_REST_Response( bms_connector_search( $data ) );
		case 'listing':
			$post = get_post( absint( $data['id'] ?? 0 ) );
			if ( ! $post || $post->post_type !== 'listings' ) {
				return new WP_REST_Response( [ 'error' => 'Esa propiedad no existe.' ], 404 );
			}
			return new WP_REST_Response( bms_connector_listing( $post, true ) );
		case 'taxonomies':
			return new WP_REST_Response( bms_connector_taxonomies() );
		case 'agents':
			return new WP_REST_Response( bms_connector_agents() );
		case 'sold':
			return bms_connector_mark_sold( $data );
		case 'leads':
			return new WP_REST_Response( bms_connector_leads_pro( $data ) );
		default:
			return new WP_REST_Response( [ 'error' => 'Accion desconocida.' ], 400 );
	}
}

// Taxonomias de Real Estate 7 que sirven como filtros.
function bms_connector_taxonomy_names(): array {
	return [ 'property_type', 'ct_status', 'beds', 'baths', 'city', 'state', 'county', 'zipcode', 'country', 'community', 'additional_features' ];
}

function bms_connector_terms( int $post_id, string $taxonomy ): array {
	$terms = get_the_terms( $post_id, $taxonomy );
	return is_array( $terms ) ? array_values( array_map( static fn( $term ) => html_entity_decode( $term->name ), $terms ) ) : [];
}

function bms_connector_meta( int $post_id, string $key ): string {
	$value = get_post_meta( $post_id, $key, true );
	return is_scalar( $value ) ? trim( (string) $value ) : '';
}

function bms_connector_user( int $user_id ): ?array {
	$user = $user_id ? get_userdata( $user_id ) : false;
	return $user ? [ 'id' => (int) $user->ID, 'name' => $user->display_name, 'email' => $user->user_email ] : null;
}

/** Propiedad con sus campos de Real Estate 7. Con $full agrega galeria, planos, archivos y toda la meta _ct_. */
function bms_connector_listing( WP_Post $post, bool $full = false ): array {
	$id = (int) $post->ID;
	$taxonomies = [];
	foreach ( bms_connector_taxonomy_names() as $taxonomy ) {
		$taxonomies[ $taxonomy ] = bms_connector_terms( $id, $taxonomy );
	}
	$brokerage = absint( bms_connector_meta( $id, '_ct_brokerage' ) );
	$listing = [
		'id'            => $id,
		'title'         => html_entity_decode( get_the_title( $post ) ),
		'url'           => get_permalink( $post ),
		'post_status'   => $post->post_status,
		'price'         => bms_connector_meta( $id, '_ct_price' ),
		'price_prefix'  => bms_connector_meta( $id, '_ct_price_prefix' ),
		'price_postfix' => bms_connector_meta( $id, '_ct_price_postfix' ),
		'mls'           => bms_connector_meta( $id, '_ct_mls' ),
		'sqft'          => bms_connector_meta( $id, '_ct_sqft' ),
		'lot_size'      => bms_connector_meta( $id, '_ct_lotsize' ),
		'year_built'    => bms_connector_meta( $id, '_ct_year_built' ),
		'parking'       => bms_connector_meta( $id, '_ct_parking' ),
		'latlng'        => bms_connector_meta( $id, '_ct_latlng' ),
		'image'         => (string) get_the_post_thumbnail_url( $post, 'large' ),
		'taxonomies'    => $taxonomies,
		'agent'         => bms_connector_user( (int) $post->post_author ),
		'co_agent'      => bms_connector_user( absint( bms_connector_meta( $id, '_ct_co_listing_agent' ) ) ),
		'brokerage'     => $brokerage ? html_entity_decode( get_the_title( $brokerage ) ) : '',
		'sold_date'     => bms_connector_meta( $id, '_ct_sold_date' ),
		'modified'      => get_post_modified_time( 'c', true, $post ),
	];
	if ( ! $full ) {
		return $listing;
	}
	$listing['description'] = wp_strip_all_tags( (string) $post->post_content );
	$listing['open_house'] = [
		'date'  => bms_connector_meta( $id, '_ct_open_house_date' ),
		'start' => bms_connector_meta( $id, '_ct_open_house_start_time' ),
		'end'   => bms_connector_meta( $id, '_ct_open_house_end_time' ),
	];
	$listing['rental'] = [
		'title'        => bms_connector_meta( $id, '_ct_rental_title' ),
		'guests'       => bms_connector_meta( $id, '_ct_rental_guests' ),
		'min_stay'     => bms_connector_meta( $id, '_ct_rental_min_stay' ),
		'checkin'      => bms_connector_meta( $id, '_ct_rental_checkin' ),
		'checkout'     => bms_connector_meta( $id, '_ct_rental_checkout' ),
		'extra_people' => bms_connector_meta( $id, '_ct_rental_extra_people' ),
		'cleaning'     => bms_connector_meta( $id, '_ct_rental_cleaning' ),
		'deposit'      => bms_connector_meta( $id, '_ct_rental_deposit' ),
		'cancellation' => bms_connector_meta( $id, '_ct_rental_cancellation' ),
	];
	$listing['video'] = bms_connector_meta( $id, '_ct_video' );
	$listing['virtual_tour'] = bms_connector_meta( $id, '_ct_virtual_tour_url' );
	$gallery = [];
	foreach ( get_attached_media( 'image', $id ) as $attachment ) {
		$gallery[] = (string) wp_get_attachment_image_url( $attachment->ID, 'large' );
	}
	$listing['gallery'] = array_values( array_filter( $gallery ) );
	// Toda la meta de Real Estate 7 (planos, archivos, IDX...) para no perder ningun campo del tema.
	$meta = [];
	foreach ( get_post_meta( $id ) as $key => $values ) {
		if ( str_starts_with( (string) $key, '_ct_' ) && isset( $values[0] ) ) {
			$value = maybe_unserialize( $values[0] );
			$meta[ $key ] = is_scalar( $value ) ? (string) $value : $value;
		}
	}
	$listing['meta'] = $meta;
	return $listing;
}

function bms_connector_search( array $data ): array {
	$per_page = min( 50, max( 1, absint( $data['per_page'] ?? 20 ) ) );
	$args = [
		'post_type'      => 'listings',
		'post_status'    => 'publish',
		'posts_per_page' => $per_page,
		'paged'          => max( 1, absint( $data['page'] ?? 1 ) ),
		'orderby'        => 'date',
		'order'          => 'DESC',
	];
	$search = sanitize_text_field( (string) ( $data['search'] ?? '' ) );
	$meta_query = [];
	if ( $search !== '' ) {
		// Busca en el titulo (la direccion) y en el MLS.
		if ( preg_match( '/^[A-Za-z0-9-]{3,}$/', $search ) ) {
			$by_mls = get_posts( [ 'post_type' => 'listings', 'post_status' => 'publish', 'fields' => 'ids', 'posts_per_page' => 20, 'meta_key' => '_ct_mls', 'meta_value' => $search ] );
		}
		if ( ! empty( $by_mls ) ) {
			$args['post__in'] = $by_mls;
		} else {
			$args['s'] = $search;
		}
	}
	$tax_query = [];
	foreach ( bms_connector_taxonomy_names() as $taxonomy ) {
		$value = sanitize_text_field( (string) ( $data['filters'][ $taxonomy ] ?? '' ) );
		if ( $value !== '' ) {
			$tax_query[] = [ 'taxonomy' => $taxonomy, 'field' => 'name', 'terms' => $value ];
		}
	}
	if ( $tax_query ) {
		$args['tax_query'] = $tax_query;
	}
	foreach ( [ 'min_price' => '>=', 'max_price' => '<=' ] as $key => $compare ) {
		$value = preg_replace( '/[^0-9.]/', '', (string) ( $data[ $key ] ?? '' ) );
		if ( $value !== '' ) {
			$meta_query[] = [ 'key' => '_ct_price', 'value' => (float) $value, 'compare' => $compare, 'type' => 'NUMERIC' ];
		}
	}
	if ( $meta_query ) {
		$args['meta_query'] = $meta_query;
	}
	$query = new WP_Query( $args );
	return [
		'total'    => (int) $query->found_posts,
		'page'     => (int) $args['paged'],
		'per_page' => $per_page,
		'items'    => array_map( static fn( $post ) => bms_connector_listing( $post ), $query->posts ),
	];
}

function bms_connector_taxonomies(): array {
	$result = [];
	foreach ( bms_connector_taxonomy_names() as $taxonomy ) {
		if ( ! taxonomy_exists( $taxonomy ) ) {
			continue;
		}
		$terms = get_terms( [ 'taxonomy' => $taxonomy, 'hide_empty' => true ] );
		$result[ $taxonomy ] = is_array( $terms ) ? array_values( array_map( static fn( $term ) => [ 'name' => html_entity_decode( $term->name ), 'count' => (int) $term->count ], $terms ) ) : [];
	}
	return $result;
}

// Agentes y brokers: autores de propiedades y usuarios con roles del tema.
function bms_connector_agents(): array {
	$users = get_users( [ 'role__in' => [ 'agent', 'broker', 'administrator', 'editor', 'author', 'contributor' ], 'number' => 500 ] );
	$agents = [];
	foreach ( $users as $user ) {
		$agents[] = [
			'id'       => (int) $user->ID,
			'name'     => $user->display_name,
			'email'    => $user->user_email,
			'roles'    => array_values( $user->roles ),
			'listings' => (int) count_user_posts( $user->ID, 'listings', true ),
		];
	}
	return $agents;
}

/** Marca la propiedad como vendida: estado del tema "Sold", fecha y precio de venta. */
function bms_connector_mark_sold( array $data ): WP_REST_Response {
	$post = get_post( absint( $data['id'] ?? 0 ) );
	if ( ! $post || $post->post_type !== 'listings' ) {
		return new WP_REST_Response( [ 'error' => 'Esa propiedad no existe.' ], 404 );
	}
	$term = get_term_by( 'slug', 'sold', 'ct_status' ) ?: get_term_by( 'name', 'Sold', 'ct_status' ) ?: get_term_by( 'name', 'Vendido', 'ct_status' );
	if ( ! $term ) {
		$created = wp_insert_term( 'Sold', 'ct_status', [ 'slug' => 'sold' ] );
		if ( is_wp_error( $created ) ) {
			return new WP_REST_Response( [ 'error' => 'No se pudo crear el estado Sold.' ], 500 );
		}
		$term = get_term( $created['term_id'], 'ct_status' );
	}
	wp_set_object_terms( $post->ID, [ (int) $term->term_id ], 'ct_status', false );
	$date = preg_match( '/^\d{4}-\d{2}-\d{2}$/', (string) ( $data['date'] ?? '' ) ) ? (string) $data['date'] : gmdate( 'Y-m-d' );
	update_post_meta( $post->ID, '_ct_sold_date', $date );
	$price = preg_replace( '/[^0-9.]/', '', (string) ( $data['price'] ?? '' ) );
	if ( $price !== '' ) {
		update_post_meta( $post->ID, '_bms_sold_price', $price );
	}
	return new WP_REST_Response( bms_connector_listing( $post ) );
}

/**
 * Envia un evento al CRM. Los fallidos quedan en cola (200 como maximo) y se reintentan cada hora.
 *
 * @return true|WP_Error
 */
function bms_connector_send( array $event, bool $blocking = true ) {
	if ( ! bms_connector_ready() ) {
		return new WP_Error( 'bms_connector', 'Configura la direccion de BMS, la empresa y el secreto.' );
	}
	$settings = bms_connector_settings();
	$event['sent_at'] = gmdate( 'c' );
	$body = wp_json_encode( $event, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES );
	$timestamp = (string) time();
	$response = wp_remote_post( $settings['bms_url'] . '/modules/crm/extensions/wordpress/' . rawurlencode( $settings['company'] ) . '/hook', [
		'timeout'     => $blocking ? 8 : 2,
		'blocking'    => $blocking,
		'redirection' => 0,
		'headers'     => [
			'Content-Type'    => 'application/json',
			'X-BMS-Timestamp' => $timestamp,
			'X-BMS-Signature' => bms_connector_sign( $timestamp, $body, $settings['secret'] ),
		],
		'body'        => $body,
	] );
	if ( ! $blocking ) {
		return true;
	}
	$status = is_wp_error( $response ) ? 0 : (int) wp_remote_retrieve_response_code( $response );
	if ( $status >= 200 && $status < 300 ) {
		return true;
	}
	if ( ( $event['event'] ?? '' ) !== 'ping' ) {
		$queue = get_option( BMS_CONNECTOR_QUEUE, [] );
		$queue = is_array( $queue ) ? $queue : [];
		$queue[] = $event;
		update_option( BMS_CONNECTOR_QUEUE, array_slice( $queue, -200 ), false );
	}
	return new WP_Error( 'bms_connector', $status === 0 ? 'No se pudo contactar BMS.' : ( $status === 401 || $status === 403 ? 'BMS rechazo la firma: revisa el secreto y el codigo de la empresa.' : 'BMS respondio HTTP ' . $status . '.' ) );
}


register_activation_hook( __FILE__, static function () {
	if ( ! wp_next_scheduled( BMS_CONNECTOR_CRON ) ) {
		wp_schedule_event( time() + 300, 'hourly', BMS_CONNECTOR_CRON );
	}
} );

register_deactivation_hook( __FILE__, static function () {
	wp_clear_scheduled_hook( BMS_CONNECTOR_CRON );
} );

add_action( BMS_CONNECTOR_CRON, static function () {
	$queue = get_option( BMS_CONNECTOR_QUEUE, [] );
	if ( ! is_array( $queue ) || ! $queue ) {
		return;
	}
	update_option( BMS_CONNECTOR_QUEUE, [], false );
	foreach ( $queue as $event ) {
		if ( is_array( $event ) ) {
			bms_connector_send( $event ); // Si vuelve a fallar se encola otra vez.
		}
	}
} );

// Propiedad publicada a la que pertenece una URL de este sitio, o null.
function bms_connector_listing_from_url( string $url ): ?array {
	if ( $url === '' || wp_parse_url( $url, PHP_URL_HOST ) !== wp_parse_url( home_url(), PHP_URL_HOST ) ) {
		return null;
	}
	$post = get_post( url_to_postid( $url ) );
	return $post && $post->post_type === 'listings' && $post->post_status === 'publish' ? bms_connector_listing( $post ) : null;
}

// Texto de los campos de un formulario, sin los tecnicos.
function bms_connector_fields_text( array $fields ): string {
	$lines = [];
	foreach ( $fields as $label => $value ) {
		if ( is_array( $value ) ) {
			$value = implode( ', ', array_map( 'strval', array_filter( $value, 'is_scalar' ) ) );
		}
		$value = trim( wp_strip_all_tags( (string) $value ) );
		if ( $value === '' || preg_match( '/(nonce|recaptcha|token|_wp_http_referer|action)$/i', (string) $label ) ) {
			continue;
		}
		$lines[] = ucfirst( str_replace( [ 'ct_field_', '_' ], [ '', ' ' ], (string) $label ) ) . ': ' . $value;
	}
	return implode( "\n", $lines );
}

// Favoritos de Real Estate Core (meta wpfp_favorites): solo las propiedades agregadas.
function bms_connector_favorites_changed( int $user_id, $old, $new ): void {
	$added = array_diff( array_map( 'intval', is_array( $new ) ? $new : [] ), array_map( 'intval', is_array( $old ) ? $old : [] ) );
	$user = get_userdata( $user_id );
	if ( ! $user || ! $added ) {
		return;
	}
	foreach ( $added as $listing_id ) {
		$post = get_post( $listing_id );
		if ( $post && $post->post_type === 'listings' ) {
			bms_connector_send( [ 'event' => 'favorite', 'name' => $user->display_name, 'email' => $user->user_email, 'listing' => bms_connector_listing( $post ) ], false );
		}
	}
}

	return $check;
}, 10, 4 );

function bms_connector_recaptcha_on(): bool {
	$settings = bms_connector_settings();
	return $settings['recaptcha_enabled'] === '1' && $settings['recaptcha_site_key'] !== '' && $settings['recaptcha_secret'] !== '';
}

// Se verifica antes que CT Forms (prioridad 1): CT Forms solo valida el nonce, que es publico.
function bms_connector_check_recaptcha(): void {
	if ( ! bms_connector_recaptcha_on() || is_user_logged_in() ) {
		return;
	}
	$settings = bms_connector_settings();
	$token = sanitize_text_field( wp_unslash( (string) ( $_POST['bms_recaptcha'] ?? '' ) ) );
	if ( $token === '' ) {
		wp_send_json_error( [ 'message' => 'No pudimos verificar el formulario. Vuelve a intentarlo.' ] );
	}
	$response = wp_remote_post( 'https://www.google.com/recaptcha/api/siteverify', [
		'timeout' => 8,
		'body'    => [ 'secret' => $settings['recaptcha_secret'], 'response' => $token, 'remoteip' => sanitize_text_field( (string) ( $_SERVER['REMOTE_ADDR'] ?? '' ) ) ],
	] );
	$result = is_wp_error( $response ) ? [] : json_decode( (string) wp_remote_retrieve_body( $response ), true );
	if ( empty( $result['success'] ) || (float) ( $result['score'] ?? 0 ) < (float) $settings['recaptcha_score'] ) {
		wp_send_json_error( [ 'message' => 'No pudimos verificar el formulario. Vuelve a intentarlo.' ] );
	}
}

add_action( 'wp_ajax_ct_form_submit', 'bms_connector_check_recaptcha', 1 );
add_action( 'wp_ajax_nopriv_ct_form_submit', 'bms_connector_check_recaptcha', 1 );

// Agrega a cada formulario un token nuevo cada 90 segundos (cada token sirve una sola vez).
add_action( 'wp_footer', static function () {
	if ( ! bms_connector_recaptcha_on() || is_user_logged_in() ) {
		return;
	}
	$key = bms_connector_settings()['recaptcha_site_key'];
	?>
	<script src="https://www.google.com/recaptcha/api.js?render=<?php echo rawurlencode( $key ); ?>" async defer></script>
	<script>
	(function () {
		var key = <?php echo wp_json_encode( $key ); ?>;
		function refresh() {
			if (!window.grecaptcha || !grecaptcha.execute) { return setTimeout(refresh, 500); }
			grecaptcha.ready(function () {
				grecaptcha.execute(key, { action: 'ct_form' }).then(function (token) {
					document.querySelectorAll('form.ct-form, form[data-form-id]').forEach(function (form) {
						var input = form.querySelector('input[name="bms_recaptcha"]');
						if (!input) { input = document.createElement('input'); input.type = 'hidden'; input.name = 'bms_recaptcha'; form.appendChild(input); }
						input.value = token;
					});
				});
			});
		}
		refresh();
		setInterval(refresh, 90000);
		document.addEventListener('submit', function () { setTimeout(refresh, 1500); }, true);
	})();
	</script>
	<?php
}, 100 );

/** Leads de CT Leads Pro por paginas: datos, etapa, agente, origen, etiquetas, notas, mensajes y propiedades. */
function bms_connector_leads_pro( array $data ): array {
	global $wpdb;
	$per_page = min( 200, max( 1, absint( $data['per_page'] ?? 200 ) ) );
	$page = max( 1, absint( $data['page'] ?? 1 ) );
	$users = get_users( [
		'meta_key'   => '_ct_lp_is_lead',
		'meta_value' => '1',
		'number'     => $per_page,
		'paged'      => $page,
		'orderby'    => 'ID',
		'order'      => 'ASC',
	] );
	if ( ! $users && $page === 1 && ! get_users( [ 'role' => 'lead', 'number' => 1 ] ) ) {
		return [ 'items' => [], 'error' => get_users( [ 'number' => 1 ] ) ? '' : 'No hay leads de CT Leads Pro en este sitio.' ];
	}
	$notes_table = $wpdb->prefix . 'ct_lp_notes';
	$messages_table = $wpdb->prefix . 'ct_lp_messages';
	$has_notes = $wpdb->get_var( $wpdb->prepare( 'SHOW TABLES LIKE %s', $notes_table ) ) === $notes_table;
	$has_messages = $wpdb->get_var( $wpdb->prepare( 'SHOW TABLES LIKE %s', $messages_table ) ) === $messages_table;
	$items = [];
	foreach ( $users as $user ) {
		$id = (int) $user->ID;
		$meta = static fn( string $key ) => is_scalar( $value = get_user_meta( $id, $key, true ) ) ? trim( (string) $value ) : '';
		$tags = get_user_meta( $id, '_ct_lp_tags', true );
		$agent = absint( $meta( '_ct_lp_assigned_agent' ) );
		$notes = [];
		if ( $has_notes ) {
			foreach ( (array) $wpdb->get_results( $wpdb->prepare( "SELECT author_id, content, created_at FROM {$notes_table} WHERE lead_id = %d ORDER BY id", $id ) ) as $note ) {
				$author = get_userdata( (int) $note->author_id );
				$notes[] = [ 'date' => (string) $note->created_at, 'author' => $author ? $author->display_name : '', 'text' => wp_strip_all_tags( (string) $note->content ) ];
			}
		}
		$messages = [];
		if ( $has_messages ) {
			foreach ( (array) $wpdb->get_results( $wpdb->prepare( "SELECT from_id, body, channel, created_at FROM {$messages_table} WHERE lead_id = %d ORDER BY id LIMIT 200", $id ) ) as $message ) {
				$from = (int) $message->from_id;
				$messages[] = [
					'date'   => (string) $message->created_at,
					'author' => $from === $id ? $user->display_name : ( ( $sender = get_userdata( $from ) ) ? $sender->display_name : '' ),
					'text'   => '[' . (string) $message->channel . '] ' . wp_strip_all_tags( (string) $message->body ),
				];
			}
		}
		$listings = [];
		foreach ( array_filter( array_map( 'absint', (array) get_user_meta( $id, '_ct_lp_interested_listing_ids', true ) ) ) as $listing_id ) {
			$post = get_post( $listing_id );
			if ( $post && $post->post_type === 'listings' ) {
				$listings[] = bms_connector_listing( $post );
			}
		}
		$items[] = [
			'id'       => $id,
			'name'     => $user->display_name,
			'email'    => str_ends_with( strtolower( $user->user_email ), '.invalid' ) ? '' : $user->user_email,
			'phone'    => $meta( '_ct_lp_phone' ),
			'stage'    => $meta( '_ct_lp_pipeline_stage' ),
			'agent'    => bms_connector_user( $agent ),
			'source'   => $meta( '_ct_lp_source' ),
			'tags'     => is_array( $tags ) ? array_values( array_map( 'strval', $tags ) ) : array_filter( array_map( 'trim', explode( ',', (string) $tags ) ) ),
			'created'  => (string) $user->user_registered,
			'notes'    => $notes,
			'messages' => $messages,
			'listings' => $listings,
		];
	}
	return [ 'items' => $items, 'page' => $page ];
}
